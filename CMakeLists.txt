cmake_minimum_required(VERSION 3.20)

# 1. Point to the vcpkg toolchain
set(CMAKE_TOOLCHAIN_FILE "C:/vcpkg/scripts/buildsystems/vcpkg.cmake" CACHE STRING "")

# 2. Define the single project name
project(protein_spatial_gnn CXX)

# 3. Set global C++20 standard
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 4. Fetch the Gemmi library
include(FetchContent)
FetchContent_Declare(
    gemmi
    GIT_REPOSITORY https://github.com/project-gemmi/gemmi.git
    GIT_TAG        v0.6.5 
) 

FetchContent_GetProperties(gemmi)
if(NOT gemmi_POPULATED)
    FetchContent_Populate(gemmi)
    add_subdirectory(${gemmi_SOURCE_DIR} ${gemmi_BINARY_DIR} EXCLUDE_FROM_ALL)
endif()

# 5. Find our external packages
find_package(Python3 COMPONENTS Interpreter Development REQUIRED)
find_package(pybind11 CONFIG REQUIRED)

# 6. Define our files
set(SOURCE_FILES
    src/protein_gnn_project.cpp
    src/protein_gnn_project.h
)

# 7. Build the Pybind11 Python extension module (THIS CREATES THE TARGET)
pybind11_add_module(spatial_engine ${SOURCE_FILES})

set_target_properties(spatial_engine PROPERTIES OUTPUT_NAME "spatial_engine")

# 8. Link the external libraries to module (THIS MUST COME AFTER TARGET CREATION)
# Note: Using Python3::Module since we used find_package(Python3 ...)
target_link_libraries(spatial_engine PRIVATE 
    pybind11::module
    Python3::Module
    gemmi::headers  
)

# 9. Keep the MSVC policies generated by the template
if (POLICY CMP0141)  
  cmake_policy(SET CMP0141 NEW)  
  set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT "$<IF:$<AND:$<C_COMPILER_ID:MSVC>,$<CXX_COMPILER_ID:MSVC>>,$<$<CONFIG:Debug,RelWithDebInfo>:EditAndContinue>,$<$<CONFIG:Debug,RelWithDebInfo>:ProgramDatabase>>")
endif()
